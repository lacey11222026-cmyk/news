Create PROCEDURE [dbo].[sp_ContentLog_SelectPagedDynamic]
	 @SelectQuery NVARCHAR(MAX) ,
    @WhereCondition NVARCHAR(MAX) ,
    @OrderByExpression NVARCHAR(250) ,
    @PageIndex INT ,
    @PageSize INT ,
    @TotalRecord INT OUTPUT
AS

SET NOCOUNT ON
SET TRANSACTION ISOLATION LEVEL READ COMMITTED

	 DECLARE @SQL NVARCHAR(3250)
        SET @SQL = ' SELECT @TotalRecordOUT = COUNT(*) FROM [dbo].[ContentLog]' ;
        IF @WhereCondition IS NOT NULL
            AND LEN(@WhereCondition) > 0 
            BEGIN
                SET @SQL = @SQL + ' WHERE ' + @WhereCondition ;
            END              
        EXEC sp_executesql @SQL, N'@TotalRecordOUT int OUTPUT',
            @TotalRecordOUT = @TotalRecord OUTPUT ;   
        --SELECT  @TotalRecord ;     

        SET @SQL = ' SELECT * FROM ( ' ;
        SET @SQL = @SQL + ' SELECT row_number() OVER ( ' ; 
        IF @OrderByExpression IS NOT NULL
            AND LEN(@OrderByExpression) > 0 
            BEGIN
                SET @SQL = @SQL + ' ORDER BY ' + @OrderByExpression ;
            END
        ELSE 
            BEGIN 
                SET @SQL = @SQL + ' ORDER BY Id DESC ' ;
            END 
        SET @SQL = @SQL + ' ) AS RowIndex , ' ; 
        IF @SelectQuery IS NOT NULL
            AND LEN(@SelectQuery) > 0 
            BEGIN
                SET @SQL = @SQL + @SelectQuery ;
            END
        ELSE 
            BEGIN
                SET @SQL = @SQL + ' * ' ;
            END
        SET @SQL = @SQL + ' FROM [dbo].[ContentLog] ' ;
        IF @WhereCondition IS NOT NULL
            AND LEN(@WhereCondition) > 0 
            BEGIN
                SET @SQL = @SQL + ' WHERE ' + @WhereCondition ;
            END           
        SET @SQL = @SQL + ' ) AS _Temp
        WHERE RowIndex BETWEEN ' + CONVERT(NVARCHAR(50), @PageSize
            * ( @PageIndex - 1 ) + 1) + ' AND '
            + CONVERT(NVARCHAR(50), @PageSize * @PageIndex) 

        PRINT @SQL ;
        EXEC sp_executesql @SQL ;       
      SELECT * FROM [dbo].[ContentLog]   
						
--endregion

CREATE PROCEDURE [dbo].[sp_NotiRead_InsertMulti]
	
	@UserName nvarchar(50),
	@ExpireDate int,
	@NotiIds nvarchar(max)
AS

SET NOCOUNT ON

INSERT INTO [dbo].[NotiRead]
           ([UserName]
           ,[NotiId]
		   ,[ExpireDate]
           ,[CreatedTime]) SELECT
						   @UserName
						   ,CONVERT(BIGINT,[Value])
						   ,@ExpireDate int
						   ,GETDATE()
					FROM [FN_Parameters_Split] (@NotiIds,';')
					WHERE [Value] IS NOT NULL AND [Value]>0
		
	
	RETURN @@IDENTITY;
